import json
from pathlib import Path
from urllib.request import Request, urlopen

depot = "Chrisnox-209/Pacman-42"
issues = []
page = 1

while True:
    url = (
        f"https://api.github.com/repos/{depot}/issues"
        f"?state=all&per_page=100&page={page}"
    )

    requete = Request(url, headers={"User-Agent": "Pacman-42"})

    with urlopen(requete) as reponse:
        resultat = json.load(reponse)

    if not resultat:
        break

    for issue in resultat:
        if "pull_request" in issue:
            continue

        issues.append({
            "number": issue["number"],
            "title": issue["title"],
            "state": issue["state"],
            "author": issue["user"]["login"],
            "assignees": [u["login"] for u in issue["assignees"]],
            "labels": [l["name"] for l in issue["labels"]],
            "body": issue["body"],
            "created_at": issue["created_at"],
            "updated_at": issue["updated_at"],
            "closed_at": issue["closed_at"],
            "comments_count": issue["comments"],
            "url": issue["html_url"],
        })

    if len(resultat) < 100:
        break

    page += 1

Path("issues.json").write_text(
    json.dumps(issues, indent=2, ensure_ascii=False),
    encoding="utf-8",
)

print(f"{len(issues)} Issues exportées dans issues.json")
PY
